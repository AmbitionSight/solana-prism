//! prism_vulnerable — the build the demo attacks.

use anchor_lang::prelude::*;

declare_id!("7SdcJctnNWx1CPrfJr5MR8jWgDLtW173FoYyCm8rP8U2");

#[program]
pub mod prism_vulnerable {
    pub fn initialize(ctx: Context<Initialize>) -> Result<()> {
        let vault = &mut ctx.accounts.vault;
        vault.authority = ctx.accounts.authority.key();
        vault.balance = 1000;
        Ok(())
    }

    pub fn withdraw(ctx: Context<Withdraw>, amount: u64) -> Result<()> {
        let vault = &mut ctx.accounts.vault;
        // The address matches, so this passes — nobody ever checked who signed.
        require!(
            vault.authority == ctx.accounts.authority.key(),
            ErrorCode::RequireViolated
        );
        vault.balance -= amount;
        Ok(())
    }
}

#[derive(Accounts)]
pub struct Withdraw<'info> {
    #[account(mut)]
    pub vault: Account<'info, Vault>,
    /// CHECK: address only — no signature required. This is the bug.
    pub authority: AccountInfo<'info>,
}

#[account]
pub struct Vault { pub authority: Pubkey, pub balance: u64 }

03 — 这个合约是干什么的

它是一个金库。

有人把钱放进去，只有这个人能取出来。合约里记着他的地址，那就是「管理员」。

正常的一次取钱

管理员
签名，调用 withdraw(500)
运行时
检查合约要求了什么签名
合约要求：管理员必须签 → 他签了 ✓
合约代码
对比地址 → 匹配 → balance -= 500
取钱成功

这一幕是整段的锚点。没有它，后面发生什么都看不懂。

04 — 拆解

把它拆成一次调用里发生的每一步。

你
withdraw(ctx, amount: 500)
你调用了它 —— 签了名，付了手续费。这一步两个版本都有，手续费总得有人付。
你
vault · authority
你把这两个账户一并交了进去。
运行时 · 声明说了算
authority: AccountInfo<'info>
这一格不是合约代码写的 —— 是运行时替它执行的。合约只负责「声明要什么」。而这行声明里，什么都没要。
合约代码
require!(vault.authority == authority.key())
比的是两个公开的值。
合约代码
vault.balance -= 500

停在第三格。 这一格是整个演示的关键：它由声明决定检查什么。声明里要了，它才查；声明里没要，它是空的 —— 这一格的空，就是漏洞。

05 — 攻击者要做什么

他只是把这个地址填了进去。

authority: ？？？ → authority: DMkwnXZ6…GU7T

这个地址，在链上人人可见。管理员的公钥不是秘密 —— 公钥从来不是秘密。

没有破解。没有伪造。没有窃取密钥。
他只是把一个公开信息，写进了那个位置。

平时需要什么才能动这个金库

  • 管理员的私钥（用来签名）

现在只需要什么

  • 管理员的地址（公开信息）

07 — 为什么会犯这个错

因为你在以太坊里这么写过，而它是对的。

Solidity · 以太坊

require(msg.sender == owner);

msg.sender 是协议根据交易签名算出来的。
协议层已经验证过签名了，所以「比对地址」就是在验证身份。

这行代码正确、充分、没问题。

Rust · Solana

require!(vault.authority ==
         ctx.accounts.authority.key());

Solana 没有 msg.sender。
这里的 authority 只是别人递进来的一个公钥。

它什么都不证明。

以太坊Solana
状态在哪在合约里面在别人递进来的账户里
调用者是谁协议告诉你不告诉你 —— 除非你声明 Signer

以太坊里，「你是谁」是协议告诉你的。
Solana 里，「你是谁」得你自己要求。

所以一个写惯 Solidity 的人，手会自己敲出那行 require! —— 它长得和 Solidity 的检查一模一样，而且看起来更稳妥。它编译通过、测试通过、评审也容易滑过去。

08 — 修复

改一行。

不是加一行检查，是改一行声明。

-     pub authority: AccountInfo<'info>,
+     pub authority: Signer<'info>,

就这。漏洞不在那行 require! 里 —— 那行本来就是对的，它确实该比对地址。

漏洞在这一行声明：它向运行时什么都没要。

两个版本，全文只差这一行。

09 — 同一笔攻击，再试一次

打修复版。

你
签名，付手续费
运行时
合约要求了哪些签名？
声明里写的：authority 必须签名
检查：管理员的签名在吗？ → 不在
交易被拒绝

合约代码一行都没有执行。 攻击者的圆点撞在第二格上就停住了 —— 那是运行时替他挡的。

两版并排

漏洞版修复版
第二格「运行时」里写着无要求authority 必须签名
交易上的签名数11（不够）
结果攻击成功交易被拒

两版的差别，只有那一格里写的东西。
